// Helper to focus the first focusable cell in a child DataTable so keyboard navigation begins there
function focusFirstCellOfChild(dt) {
    if (!dt || typeof dt.cell !== 'function') return false;

    var cell = dt.cell(0, 0);
    if (!cell || typeof cell.node !== 'function') return false;

    var node = cell.node();
    if (!node) return false;

    var $node = $(node);
    $node.attr('tabindex', 0);
    $node.focus();

    // Update KeyTable's internal focus if available (feature-detect)
    if (typeof cell.focus === 'function') {
        cell.focus();
        return true;
    }

    if (dt.keys && typeof dt.keys === 'function') {
        var kt = dt.keys();
        if (kt && typeof kt.focus === 'function') {
            var idx = (typeof cell.index === 'function') ? cell.index() : null;
            kt.focus(idx || node);
            return true;
        }
    }

    return true;
}

function findElement(arr, propName, propValue) {
    // Defensive checks
    if (!Array.isArray(arr) || arr.length === 0 || propValue == null) return null;

    var pv = String(propValue).trim().toLowerCase();

    // Use Array.prototype.find for clarity and reliability
    var found = arr.find(function (item) {
        if (!item || !(propName in item)) return false;
        var val = item[propName];
        if (val == null) return false;
        return String(val).trim().toLowerCase() === pv;
    });

    return found || null;
}

/**
 * Formats a datetime as mm-dd-yyyy
 * @param {string} date value to be formatted
 * @returns {string} formatted value
 */
function formatDate(date) {
    return isIsoDateRegex(date) && !isUndefinedOrEmpty(date) ? date.substring(5, 7) + '/' + date.substring(8, 10) + '/' + date.substring(0, 4) : date;

//check for ISO date string only
function isIsoDateRegex(dateToValidate) {
    var regex = /^(\d{4})-0?(\d+)-0?(\d+)[T ]0?(\d+):0?(\d+):0?(\d+)$/;
    if (regex.test(dateToValidate)) {
        return true;
    }
    return false;
}

/**
 * Formats a date with the specified format
 * @param {any} date value to be formatted
 * @param {any} format format to use
 * Available placeholders:
 *      mm : month (1-2 digits)
 *      MM : month (2 digits)
 *      MMM : month name (abbreviated)
 *      MMMM : month name (full)
 *      dd : day of the month (1-2 digits)
 *      DD  : day of the month (2 digits)
 *      ddd : day name (abbreviated)
 *      dddd : day name (full)
 *      yy : year (2 digits)
 *      yyyy : year (4 digits)
 *      hr : hour (2 digits)
 *      HR : hour (12hr/2 digits)
 *      mn : minutes (2 digits)
 *      ss : seconds (2 digits)
 *      ms : milliseconds (3 digits)
 *      ap : AM/PM
 * @returns {string} formatted value
 */
function formatDate2(date, format) {
    if (isUndefinedOrEmpty(date)) {
        return date;
    }
    var dateObject = undefined;
    switch (typeof date) {
        case 'object':
            if (date instanceof Date) {
                dateObject = date;
            }
            break;
        case 'string':
            var dateString = date;
            if (!isIsoDateRegex(dateString)) {
                dateString = date.toString().indexOf('T') === -1 ? date.toString() + 'T00:00:00' : date.toString();
            }
            dateObject = new Date(dateString);
            break;
        case 'number':
            // assume milliseconds since 1/1/1970
            dateObject = new Date(date);
            break;
    }
    if (!isUndefinedOrEmpty(dateObject)) {
        var month = dateObject.getMonth() + 1;
        var day = dateObject.getDate();
        var year = dateObject.getFullYear();

        return format
            .replace('mm', month)
            .replace('MMMM', dateObject.toLocaleDateString('en', { month: 'long' }))
            .replace('MMM', dateObject.toLocaleDateString('en', { month: 'short' }))
            .replace('MM', ('0' + month).slice(-2))
            .replace('dddd', dateObject.toLocaleDateString('en', { weekday: 'short' }))
            .replace('ddd', dateObject.toLocaleDateString('en', { weekday: 'long' }))
            .replace('dd', day)
            .replace('DD', ('0' + day).slice(-2))
            .replace('yyyy', year)
            .replace('yy', year.toString().slice(-2))
            .replace('hr', ('0' + dateObject.getHours()).slice(-2))
            .replace('HR', ('0' + ((dateObject.getHours() + 11) % 12 + 1)).slice(-2))
            .replace('mn', ('0' + dateObject.getMinutes()).slice(-2))
            .replace('ss', ('0' + dateObject.getSeconds()).slice(-2))
            .replace('ms', ('00' + dateObject.getMilliseconds()).slice(-3))
            .replace('ap', dateObject.getHours() < 12 ? 'AM' : 'PM')
            ;
    }
    return undefined;
}

function errorMessageShow(control, message) {
    if (!isUndefinedOrEmpty(message)) {

/**
 * Displays a modal confirmation dialog box
 * @param {string} title text to display in the pop-up title bar
 * @param {string} message message to be displayed in the pop-up (supports html markup)
 * @param {Function} confirmFunction callback function to execute when the confirm button is clicked
 * @param {string} alternateButtonText used to override the default button text of 'Ok'
 * @param {string} [icon] icon to be displayed
 */
function showConfirmDialog(title, message, confirmFunction, alternateButtonText, icon) {
    return new Promise((resolve) => {
        var buttonText = isUndefinedOrEmpty(alternateButtonText) ? "Ok" : alternateButtonText;
        var dialog = $(ELEMENT.DIV).html(message).dialog({
            dialogClass: 'no-close',
            title: title,
            autoOpen: true,
            modal: true,
            resizable: false,
            closeOnEscape: false,
            width: "auto",
            buttons: [
                {
                    text: buttonText,
                    class: CSS_RULE.BUTTON,
                    click: function () {
                        if (confirmFunction) {
                            confirmFunction.call();
                        }
                        $(this).dialog("destroy");
                        resolve(true);
                    }
                }
            ]
        });
        if (!isUndefinedOrEmpty(icon)) {
            let container = $(ELEMENT.DIV).addClass("ui-dialog-icon").insertBefore(dialog);
            $(ELEMENT.SPAN).addClass("ui-icon ui-icon-large" + icon).prependTo(container);
        }
    });
}

/**
 * Displays a Yes/No modal dialog box
 * @param {string} title text to display in the pop-up title bar
 * @param {string} message message to be displayed in the pop-up (supports html markup)
 * @param {Function} yesFunction function to execute when 'yes' is clicked
 * @param {Function} noFunction function to execute when 'no' is clicked
  * @param {string} [icon] icon to be displayed
*/
function showYesNoDialog(title, message, yesFunction, noFunction, icon) {
    return new Promise((resolve, reject) => {
        var dialog = $(ELEMENT.DIV).html(message).dialog({
            dialogClass: 'no-close',
            title: title,
            autoOpen: true,
            modal: true,
            resizable: false,
            closeOnEscape: false,
            buttons: [
                {
                    text: "Yes",
                    click: function () {
                        if (yesFunction && typeof yesFunction === 'function') {
                            yesFunction.call();
                        }
                        $(this).dialog("destroy");
                        resolve(true);
                    }
                },
                {
                    text: "No",
                    click: function (e) {
                        if (noFunction && typeof noFunction === 'function') {
                            noFunction.call();
                        }
                        $(this).dialog("destroy");
                        resolve(false);
                    }
                }
            ],
            open: (event) => {
                $(event.target).parent().find('.ui-dialog-buttonpane .ui-dialog-buttonset button:eq(1)').focus();
            }
        });
        if (!isUndefinedOrEmpty(icon)) {
            let container = $(ELEMENT.DIV).addClass("ui-dialog-icon").insertBefore(dialog);
            $(ELEMENT.SPAN).addClass("ui-icon ui-icon-large" + icon).prependTo(container);
        }
    });
}

/**
 * Find an object in an array
 * 
 * @param {Array} array array of objects
 * @param {string} propertyName name of the property
 * @param {any} value value to be compared
 * @returns {object} the matching object or 'undefined'
 */
function findObjectInArray(array, propertyName, value) {
    return array[getObjectIndex(array, propertyName, value)];
}

/**
 * Get the index of an object from an array
 * 
 * @param {Array} array array of objects
 * @param {string} propertyName name of the property
 * @param {any} value value to be compared
 * @returns {int} index of the matching object or -1 if not found
 */
function getObjectIndex(array, propertyName, value) {
    for (var i = 0; i < array.length; i++) {
        if (array[i] !== undefined && array[i][propertyName] === value) {
            return i;
        }
    }
    return -1;
}

/**
 * Remove an object from the specified array that matches the given property name and value
 *
 * @param {Array} array array of objects
 * @param {string} propertyName name of the property
 * @param {any} value value to be compared
 */
function removeObjectInArray(array, propertyName, value) {
    var index = getObjectIndex(array, propertyName, value);
    if (index > -1) {
        array.splice(index, 1);
    }
}

function _instanceof(left, right) {
    if (right !== null && typeof Symbol !== "undefined" && right[Symbol.hasInstance]) {
        return right[Symbol.hasInstance](left);
    } else {
        return left instanceof right;
    }
}

function _classCallCheck(instance, varructor) {
    if (!_instanceof(instance, varructor)) {
        throw new TypeError("Cannot call a class as a function");
    }
}
